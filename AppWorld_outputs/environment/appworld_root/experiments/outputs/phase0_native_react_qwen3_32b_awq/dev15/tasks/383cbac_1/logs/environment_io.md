from datetime import datetime, timedelta

# Get today's date
today = datetime.now()

# Calculate yesterday's date
yesterday_start = today - timedelta(days=1)
yesterday_start = yesterday_start.replace(hour=0, minute=0, second=0, microsecond=0)
yesterday_end = yesterday_start + timedelta(days=1) - timedelta(seconds=1)

# Format the date range for the API
yesterday_start_str = yesterday_start.strftime("%Y-%m-%d")
yesterday_end_str = yesterday_end.strftime("%Y-%m-%d")

# Retrieve Venmo transactions from yesterday
venmo_transactions = []
page_index = 0
while True:
    transactions_page = apis.venmo.show_transactions(
        access_token=venmo_access_token,
        min_created_at=yesterday_start_str,
        max_created_at=yesterday_end_str,
        page_index=page_index
    )
    if not transactions_page:
        break
    venmo_transactions.extend(transactions_page)
    page_index += 1

print(venmo_transactions)
